\cvsection{Research Systems \& Open-Source Engineering}

\textbf{TimeFlowCodec (TFC)} $\cdot$ \textit{Lead Architect \& Creator} \hfill \href{https://github.com/rchau8502/TimeFlowCodec}{github.com/rchau8502/TimeFlowCodec} \\
\textit{An orthogonal, patent-free video coding standard engineered for screen, terminal, and slide content.}
\begin{itemize}
    \setlength{\itemsep}{1pt}
    \setlength{\parskip}{0pt}
    \setlength{\parsep}{0pt}
    \item Formulated a novel video coding paradigm that discards 2D spatial DCT block transforms in favor of 1D continuous per-pixel Ordinary Least Squares (OLS) temporal regression with native 4:4:4 RGB representation.
    \item Eliminated high-frequency Gibbs ringing artifacts ($>10\%$ overshoot) and chroma bleeding around text edges.
    \item Achieved an average \textbf{$-26.8\%$ BD-Rate reduction} over H.264 (up to \textbf{$-41.2\%$} on slide decks) evaluated under standard ITU-T VCEG-M33 Common Test Conditions on JVET Screen Content Coding sequences.
    \item Designed a Leaky-Bucket Rate Controller (CRF 0--63, CBR, VBR) and Lagrangian Rate-Distortion Optimization ($J = D + \lambda R$).
\end{itemize}

\vspace{3pt}

\textbf{libtfc: Standalone ANSI C Reference Decoder} $\cdot$ \textit{Lead Developer} \hfill 2026 \\
\textit{Ultra-lightweight, zero-dependency C99 decoder engine.}
\begin{itemize}
    \setlength{\itemsep}{1pt}
    \setlength{\parskip}{0pt}
    \setlength{\parsep}{0pt}
    \item Engineered an ANSI C99 library and standalone CLI player with zero runtime dependencies.
    \item Delivered \textbf{$>840$ FPS} decode throughput ($>2{,}200$ MB/s raw RGB24 at 720p) and $>384$ FPS at 1080p, utilizing an $\mathcal{O}(1)$ sequential line-buffer without requiring multi-frame Decoded Picture Buffers (DPB).
    \item Demonstrated a \textbf{$75\%$ reduction in package power dissipation} ($0.85$ W vs. $3.42$ W) over software AV1 via macOS \texttt{powermetrics}.
\end{itemize}
